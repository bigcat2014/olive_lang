\section{Ambiguity}
\paragraph{
    Ambiguity is recognized between some of the regexes listed below.
    This ambiguity is solved in the lexer by taking the longest match or through regex precedence.
    The longest match is to pick the regex that matches the most characters (greedy matching),
    while the precedence resolution is to define a precedence of regexes such that in the case where
    two regexes match the same span one is preferred.
}
\subsection{Examples}
\paragraph{Some examples of ambiguous matches include:}
\begin{itemize}
    \item "else" matches both \begin{math}<else>\end{math} and \begin{math}<ident>\end{math}. Solved with precedence.
    \item "1.2e3" matches as either \begin{math}<float\_const>\end{math} followed by an \begin{math}<ident>\end{math} or as a \begin{math}<scientific>\end{math}. Solved with longest match.
    \item "r\textbackslash"abc\textbackslash"" matches as either an \begin{math}<ident>\end{math} followed by a \begin{math}<string\_literal>\end{math} or as just a \begin{math}<string\_literal>\end{math}. Solved with longest match.
\end{itemize}
\section{Regex}
\subsection{Substitutions}
\begin{align}
    &L &&:= &&[\text{a}-\text{z}]                     \\
    &U &&:= &&[\text{A}-\text{Z}]                     \\
    &D &&:= &&[\text{0}-\text{9}]                     \\
    &H &&:= &&[\text{a}-\text{fA}-\text{F0}-\text{9}] \\
    &O &&:= &&[\text{0}-\text{7}]                     \\
    &B &&:= &&[\text{01}]
\end{align}

\subsection{Identifiers, Literals, and Comments}
\begin{align}
    &\_\{0, 2\}L(?:L|U|D|\_)*                                                                &&\to \quad\text{IDENT}                 \\
    &U(?:L|U|D)*                                                                             &&\to \quad\text{TYPE\_IDENT}           \\
    &[\backslash+-]?D+                                                                       &&\to \begin{aligned}[t]
                                                                                                   \quad\text{NUMERIC\_LITERAL}      \\
                                                                                                   \text{(INT\_LITERAL)}             \\
                                                                                                   \text{Integer}
                                                                                                   \end{aligned}                     \\
    &\text{0}[\text{xX}](H\{1,16\})                                                          &&\to \begin{aligned}[t]
                                                                                                   \quad\text{NUMERIC\_LITERAL}      \\
                                                                                                   \text{(RAW\_BITS)}                \\
                                                                                                   \text{Hex}
                                                                                                   \end{aligned}                     \\
    &\text{0}[\text{oO}](1O\{21\}|O\{1,21\})                                                 &&\to \begin{aligned}[t]
                                                                                                   \quad\text{NUMERIC\_LITERAL}      \\
                                                                                                   \text{(RAW\_BITS)}                \\
                                                                                                   \text{Octal}
                                                                                                   \end{aligned}                     \\
    &\text{0}[\text{bB}](B\{1,64\})                                                          &&\to \begin{aligned}[t]
                                                                                                   \quad\text{NUMERIC\_LITERAL}      \\
                                                                                                   \text{(RAW\_BITS)}                \\
                                                                                                   \text{Binary}
                                                                                                   \end{aligned}                     \\
    &[\backslash+-]?(?:D+\backslash.D*|\backslash.D+)                                        &&\to \begin{aligned}[t]
                                                                                                   \quad\text{NUMERIC\_LITERAL}      \\
                                                                                                   \text{(FLOAT\_LITERAL)}           \\
                                                                                                   \text{Float}
                                                                                                   \end{aligned}                     \\
    &([\backslash+-]?(?:D+(?:\backslash.D*)?|\backslash.D+))[eE]([\backslash+-]?D+)          &&\to \begin{aligned}[t]
                                                                                                   \quad\text{NUMERIC\_LITERAL}      \\
                                                                                                   \text{(FLOAT\_LITERAL)}           \\
                                                                                                   \text{Scientific}
                                                                                                   \end{aligned}                     \\
    &(\text{r}?)"((?:[\enspace\hat{}\enspace"\backslash\backslash]|\backslash\backslash")*)" &&\to \quad\text{STRING\_LITERAL}       \\
    &\#.*\$                                                                                  &&\to \quad\text{COMMENT}
\end{align}
