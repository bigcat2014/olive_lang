\section{Backus-Naur Form}
\begin{align}
    &<ident> &&::= && <lower> \enspace <ident?> \enspace \\
    &&&&&| \enspace "\_" \enspace <lower> \enspace <ident?> \\
    &&&&&| \enspace "\_" \enspace "\_" \enspace <lower> \enspace <ident?> \\
    &<ident?> &&::= &&<lower> \enspace <ident?> \\
    &&&&&|<upper> \enspace <ident?> \\
    &&&&&|<digit> \enspace <ident?> \\
    &&&&&|\enspace "\_" \enspace <ident?> \\
    &&&&&| \enspace \varepsilon \\
    &<type\_ident> &&::= &&<upper> <type\_ident?> \\
    &<type\_ident?> &&::= &&<lower> \enspace <type\_ident?> \\
    &&&&&|<upper> \enspace <type\_ident?> \\
    &&&&&|<digit> \enspace <type\_ident?> \\
    &&&&&| \enspace \varepsilon \\
    &<float\_const> &&::= &&<mantissa> \\
    &&&&&| <hex\_const> \\
    &&&&&| <octal\_const> \\
    &&&&&| <binary\_const> \\
    &&&&&| <scientific> \\
    &<integer\_const> &&::= &&<decimal\_const> \\
    &&&&&| <hex\_const> \\
    &&&&&| <octal\_const> \\
    &&&&&| <binary\_const> \\
    &<comment> &&::=  && \enspace "\#" <comment\_text> \\
    &<comment\_text> &&::= && <comment\_char> <comment\_text> \\
    &<comment\_char> &&::= &&<lower> | <upper | <digit> | <symbol> | <non\_newline> \\
    &<string\_literal> &&::= &&<r?> <quote> <content> <quote> \\
    &<content> &&::= &&<char> <content> | \enspace \varepsilon \\
    &<char> &&::= && <escaped\_quote> | <normal\_symbol> \\
    &<scientific> &&::= &&<sign?> <mantissa> <exponent> \\
    &<mantissa> &&::= &&<digits> \enspace "." \enspace <digits?> \\
    &&&&&| \enspace "." \enspace <digits> \\
    &&&&&| <digits> \\
    &<exponent> &&::= &&"e" \enspace <decimal\_const> \\
    &<decimal\_const> &&::= &&<sign?> <digits> \\
    &<hex\_const> &&:= &&\enspace "0" \enspace "x" <hs> \\
    &&&&&| \enspace "0" \enspace "X" <hs> \\
    &<octal\_const> &&::= &&\enspace "0" \enspace "o" <os> \\
    &&&&&| \enspace "0" \enspace "O" <os> \\
    &<binary\_const> &&::= &&\enspace "0" \enspace "b" <bs> \\
    &&&&&| \enspace "0" \enspace "B" <bs> \\
\end{align}
